        function initYearMonthSelectors() {

            var records = globalDashboardData.fullMonthlyRecords;

            var yearSelect = document.getElementById("filterYear");

            var years = Object.keys(records).sort().reverse();

            var yearHtml = "";

            years.forEach(function(y) { yearHtml += `<option value="${y}">${y}년</option>`; });

            yearSelect.innerHTML = yearHtml;

            yearSelect.value = globalDashboardData.latest.defaultYear;

            updateMonthSelector(globalDashboardData.latest.defaultYear, globalDashboardData.latest.defaultMonth);

        }


        function updateMonthSelector(selectedYear, defaultMonthToSet) {

            var months = Object.keys(globalDashboardData.fullMonthlyRecords[selectedYear]).sort().reverse();

            var monthSelect = document.getElementById("filterMonth");

            var monthHtml = "";

            months.forEach(function(m) { monthHtml += `<option value="${m}">${m}월</option>`; });

            monthSelect.innerHTML = monthHtml;

            if (defaultMonthToSet && months.indexOf(defaultMonthToSet) !== -1) { monthSelect.value = defaultMonthToSet; }

            applySelectedSnapshot(selectedYear, monthSelect.value);

        }


        function onYearSelectorChange(year) { updateMonthSelector(year, null); }

        function onMonthSelectorChange(month) { var year = document.getElementById("filterYear").value; applySelectedSnapshot(year, month); }


        function applySelectedSnapshot(year, month) {

            var targetData = globalDashboardData.fullMonthlyRecords[year][month];

            diffEl.innerText = (targetData.diffRaw >= 0 ? "▲ +" : "▼ ") + targetData.diffRaw.toLocaleString() + "원";

            diffEl.className = targetData.diffRaw >= 0 ? "text-[10px] text-emerald-600 font-bold mt-0.5" : "text-[10px] text-rose-600 font-bold mt-0.5";

            document.getElementById("tblEstate").innerText = targetData.estate.toLocaleString() + " 원";

            document.getElementById("tblInvest").innerText = targetData.invest.toLocaleString() + " 원";

            document.getElementById("tblDeposit").innerText = targetData.deposit.toLocaleString() + " 원";

            document.getElementById("tblRetire").innerText = targetData.retirementAmt.toLocaleString() + " 원";

            updatePieChartInstance(targetData.estate, targetData.invest, targetData.deposit, targetData.retirementAmt);

        }


        function initGangrokSelectors() {

            var grData = globalDashboardData.gangrokMonthly;

            var grYearSelect = document.getElementById("grYear");

            var years = Object.keys(grData).sort().reverse();

            var yearHtml = "";

            years.forEach(function(y) { yearHtml += `<option value="${y}">${y}년</option>`; });

            grYearSelect.innerHTML = yearHtml;

            grYearSelect.value = globalDashboardData.grLatest.year || years[0];

            updateGangrokMonthSelector(grYearSelect.value, globalDashboardData.grLatest.month);

        }


        function updateGangrokMonthSelector(selectedYear, defaultMonthToSet) {

            var months = Object.keys(globalDashboardData.gangrokMonthly[selectedYear]).sort().reverse();

            var grMonthSelect = document.getElementById("grMonth");

            var monthHtml = "";

            months.forEach(function(m) { monthHtml += `<option value="${m}">${m}월</option>`; });

            grMonthSelect.innerHTML = monthHtml;

            if (defaultMonthToSet && months.indexOf(defaultMonthToSet) !== -1) { grMonthSelect.value = defaultMonthToSet; }

            applyGangrokSnapshot(selectedYear, grMonthSelect.value);

        }


        function onGrYearChange(year) { updateGangrokMonthSelector(year, null); }

        function onGrMonthChange(month) { var year = document.getElementById("grYear").value; applyGangrokSnapshot(year, month); }


        function applyGangrokSnapshot(year, month) {

            var target = globalDashboardData.gangrokMonthly[year][month];

            if(!target) return;

            document.getElementById("tblGrDeposit").innerText = target.cash.toLocaleString() + " 원";

            document.getElementById("tblGrStock").innerText = target.stock.toLocaleString() + " 원";

            document.getElementById("tblGrTotal").innerText = target.total.toLocaleString() + " 원";


            if(!gangrokPieChart) {

                gangrokPieChart = new ApexCharts(document.querySelector("#gangrokPieChart"), {

                    series: [target.cash, target.stock],

                    chart: { type: 'pie', width: '100%', height: 180, sparkline: { enabled: true } },

                    labels: ['강록 예금', '강록 주식'],

                    colors: ['#a855f7', '#ec4899'],

                    dataLabels: { enabled: true, formatter: function (val) { return val.toFixed(1) + "%"; } }

                });

                gangrokPieChart.render();

            } else {

                gangrokPieChart.updateSeries([target.cash, target.stock]);

            }

        }


        function formatAssetUnit(value) {

            if (value >= 10000) { return (value / 10000).toFixed(1) + "억"; }

            return value.toLocaleString() + "만";

        }


        function renderPieChart(estate, invest, deposit, retire) {

            pieChartObject = new ApexCharts(document.querySelector("#pieChart"), {

                series: [estate, invest, deposit, retire],

                chart: { type: 'pie', width: '100%', height: 180, sparkline: { enabled: true } },

                labels: ['부동산', '주식/코인', '예적금', '노후자금'],

                legend: { show: false },

            });

            pieChartObject.render();

        }


        function updatePieChartInstance(estate, invest, deposit, retire) {

            if(!pieChartObject) { renderPieChart(estate, invest, deposit, retire); } 

            else { pieChartObject.updateSeries([estate, invest, deposit, retire]); }

        }
